def _svg_data_uri(path: Path) -> str:
    try:
        data = path.read_bytes()
        return "data:image/svg+xml;base64," + base64.b64encode(data).decode()
    except Exception:
        return ""

def _load_query_state():
    params = st.query_params
    sb = params.get("sb")
    if sb in {"chat", "overview", "data", "settings"}:
        st.session_state.sb_nav = sb
    theme = params.get("theme")
    if theme in {"Light", "Dark"}:
        st.session_state.ui_theme = theme
    try:
        k = int(params.get("topk", 0))
        if 1 <= k <= 50:
            st.session_state.rag_top_k = k
    except Exception:
        pass
    try:
        n = int(params.get("topn", 0))
        if 1 <= n <= 20:
            st.session_state.rag_top_n = n
    except Exception:
        pass
    start = params.get("start")
    end = params.get("end")
    try:
        if start:
            st.session_state.flt_start = datetime.fromisoformat(start).date()
        if end:
            st.session_state.flt_end = datetime.fromisoformat(end).date()
    except Exception:
        pass
    cats = params.get("cats")
    if cats:
        st.session_state.flt_cats = cats.split(",")
    st.session_state.qs_loaded = True


def _sync_query_state():
    st.query_params.update(
        sb=st.session_state.sb_nav,
        theme=st.session_state.ui_theme,
        topk=st.session_state.rag_top_k,
        topn=st.session_state.rag_top_n,
        start=str(st.session_state.flt_start) if st.session_state.flt_start else "",
        end=str(st.session_state.flt_end) if st.session_state.flt_end else "",
        cats=",".join(st.session_state.flt_cats) if st.session_state.flt_cats else "",
    )


# --- Background helpers ---
@st.cache_data
def get_blurred_background(image_path, blur_radius=10):
    try:
        img = Image.open(image_path)
        blurred_img = img.filter(ImageFilter.GaussianBlur(blur_radius))
        buffered = io.BytesIO()
        blurred_img.save(buffered, format="PNG")
        img_str = base64.b64encode(buffered.getvalue()).decode()
        return img_str
    except Exception:
        return None
